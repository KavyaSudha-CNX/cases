-- Category
INSERT INTO master_category (category_name)
SELECT DISTINCT category_name
FROM tmp_service_home_charger
WHERE category_name IS NOT NULL
ON CONFLICT (category_name) DO NOTHING;

-- Primary Category
INSERT INTO master_primary_category (primary_category_name)
SELECT DISTINCT primary_category_name
FROM tmp_service_home_charger
WHERE primary_category_name IS NOT NULL
ON CONFLICT (primary_category_name) DO NOTHING;


-- Secondary Category
INSERT INTO master_secondary_category (secondary_category_name)
SELECT DISTINCT secondary_category_name
FROM tmp_service_home_charger
WHERE secondary_category_name IS NOT NULL
ON CONFLICT (secondary_category_name) DO NOTHING;


-- Tertiary Category
INSERT INTO master_tertiary_category (tertiary_category_name)
SELECT DISTINCT tertiary_category_name
FROM tmp_service_home_charger
WHERE tertiary_category_name IS NOT NULL
ON CONFLICT (tertiary_category_name) DO NOTHING;

-- Mapping
INSERT INTO master_category_mapping
(
    lob_channel_case_type_id,
    category_id,
    primary_category_id,
    secondary_category_id,
    tertiary_category_id
)
SELECT DISTINCT
    d.lob_channel_case_type_id,
    c.id,
    p.id,
    s.id,
    t.id
FROM tmp_service_home_charger d

JOIN master_category c
    ON LOWER(TRIM(c.category_name)) =
       LOWER(TRIM(d.category_name))

JOIN master_primary_category p
    ON LOWER(TRIM(p.primary_category_name)) =
       LOWER(TRIM(d.primary_category_name))

JOIN master_secondary_category s
    ON LOWER(TRIM(s.secondary_category_name)) =
       LOWER(TRIM(d.secondary_category_name))

JOIN master_tertiary_category t
    ON LOWER(TRIM(t.tertiary_category_name)) =
       LOWER(TRIM(d.tertiary_category_name))

ON CONFLICT DO NOTHING;
